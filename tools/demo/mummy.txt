# Starlace Bandage: a row of mummies of many model shapes (summoned already wrapped through their attachment), then a
# pig wrapped by hand with a bandage, the row at night (glowing wraps), and the pig unwrapped with shears.
world lacuna_mummy flat
wait 60
hud off
cmd time set 6000
cmd gamerule doDaylightCycle false
cmd gamerule doMobSpawning false
cmd gamerule sendCommandFeedback false
cmd give @s lacuna:starlace_bandage 16
cmd give @s minecraft:shears
cmd summon minecraft:zombie -8 -60 6 {"neoforge:attachments":{"lacuna:mummy":{wraps:2,needed:2}},NoAI:1b,Silent:1b,Rotation:[180f,0f]}
cmd summon minecraft:skeleton -6 -60 6 {"neoforge:attachments":{"lacuna:mummy":{wraps:2,needed:2}},NoAI:1b,Silent:1b,Rotation:[180f,0f]}
cmd summon minecraft:creeper -4 -60 6 {"neoforge:attachments":{"lacuna:mummy":{wraps:2,needed:2}},NoAI:1b,Silent:1b,Rotation:[180f,0f]}
cmd summon minecraft:villager -2 -60 7 {"neoforge:attachments":{"lacuna:mummy":{wraps:1,needed:1}},NoAI:1b,Silent:1b,Rotation:[180f,0f]}
cmd summon minecraft:iron_golem 0.5 -60 9 {"neoforge:attachments":{"lacuna:mummy":{wraps:2,needed:2}},NoAI:1b,Silent:1b,Rotation:[180f,0f]}
cmd summon minecraft:cow 3 -60 7 {"neoforge:attachments":{"lacuna:mummy":{wraps:1,needed:1}},NoAI:1b,Silent:1b,Rotation:[150f,0f]}
cmd summon minecraft:sheep 5.5 -60 6 {"neoforge:attachments":{"lacuna:mummy":{wraps:1,needed:1}},NoAI:1b,Silent:1b,Rotation:[150f,0f]}
cmd summon minecraft:spider 8 -60 6 {"neoforge:attachments":{"lacuna:mummy":{wraps:2,needed:2}},NoAI:1b,Silent:1b,Rotation:[160f,0f]}
cmd summon minecraft:pig 0.5 -60 2 {NoAI:1b,Rotation:[150f,0f]}
cmd tp @s 0.5 -60 0 facing 0.5 -59.6 2
wait 60
select 0
use on
wait 3
use off
wait 10
shot mummy_row_day
cmd tp @s -6 -60 2.5 facing -6 -59 6
wait 20
shot mummy_closeup
cmd tp @s 0.5 -60 0 facing 0.5 -59.6 2
cmd time set 18000
wait 30
shot mummy_row_night
cmd time set 6000
select 1
wait 5
use on
wait 3
use off
wait 20
shot mummy_unwrapped
quit
