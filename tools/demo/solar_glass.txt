# Divine Magnifying Glass: client shader beam by day (yellow), by night (white) and underwater (blue), first person
# (beam leaves the drawn lens) and third person (front camera: the beam comes toward the viewer).
world lacuna_glass flat
wait 60
hud off
cmd gamerule sendCommandFeedback false
cmd time set 6000
cmd give @s lacuna:divine_magnifying_glass
select 0
cmd fill ~-2 ~ ~7 ~2 ~2 ~7 minecraft:stone_bricks
cmd setblock ~ ~ ~6 minecraft:oak_planks
look 0 12
wait 20
hud on
clearchat
use on
wait 24
shot glass_day_first
hud off
camera front
wait 4
shot glass_day_front
camera first
use off
cmd time set 18000
wait 15
hud on
clearchat
use on
wait 24
shot glass_night_first
hud off
camera front
wait 4
shot glass_night_front
camera first
use off
cmd time set 6000
cmd fill ~-5 ~ ~-3 ~5 ~3 ~6 minecraft:water replace air
wait 20
hud on
clearchat
use on
wait 24
shot glass_underwater_first
hud off
camera front
wait 4
shot glass_underwater_front
camera first
use off
quit
