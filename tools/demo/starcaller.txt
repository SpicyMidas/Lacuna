# Starcaller Trumpet: toot pose, then a star called down on a pig: it twinkles into being, falls across the night sky
# with its trail, strikes, and leaves a Fallen Star (close-up at the end).
world lacuna_star flat
wait 60
hud off
cmd time set 14000
cmd gamerule doDaylightCycle false
cmd gamerule sendCommandFeedback false
cmd give @s lacuna:starcaller_trumpet
cmd summon minecraft:pig 0 -60 14 {NoAI:1b}
cmd tp @s 0 -60 -20
wait 40
cmd tp @s 0 -60 -20 facing 0 -59.4 14
wait 5
select 0
use on
wait 12
shot star_toot
use off
look 0 -30
wait 2
shot star_appear
wait 12
shot star_fall1
wait 14
shot star_fall2
wait 12
shot star_fall3
look 0 4
wait 14
shot star_impact
wait 30
shot star_after
cmd tp @s 0 -58.5 9 facing 0 -60 14
wait 20
shot star_block
quit
